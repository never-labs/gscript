// generic_ai_evaluation.source.txt - Provider-free AI evaluation dialect specimen.
//
// Tests materialize this fixture as a temporary .leia file before invoking
// `leia evaluate`, so the harness can validate the dialect without registering
// another FinRobot product example.

evaluate "generic ai evaluation dialect specimen" {
    rows := eval.load_jsonl("generic_ai_evaluation_dataset.jsonl")
    eval.fail_if(#rows != 2, "dataset manifest fixture must contain two rows")

    for _, row := range ipairs(rows) {
        eval.case(row.id, func() {
            eval.metric("dataset_item_valid", row.input != "")
            eval.metric("answer_chars", #row.answer)
            eval.metric("rubric_label", row.rubric_label)

            if row.needs_judge {
                result, err := eval.judge({
                    model: "mock-ai-eval-judge"
                    messages: {
                        llm.system("Score whether the answer satisfies the rubric. Return pass or fail."),
                        llm.user("Rubric: " .. row.rubric .. " Answer: " .. row.answer),
                    }
                })
                assert(err == nil)
                assert(result.text == row.expected_judgment)
                eval.metric("judge_passed", result.text == row.expected_judgment)
            } else {
                eval.metric("judge_passed", true)
            }
        })
    }

    usage := eval.usage()
    assert(usage.turns == 1)
    assert(usage.tokens == 7)
    eval.budget({turns: 1, tokens: 16, cost: 0.02})
}
