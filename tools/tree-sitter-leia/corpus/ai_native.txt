models {
  default: "demo"
  demo: { provider_model: "mock" }
}

tool lookup(query) {
  return "doc:" .. query, nil
}

agent defaults {
  model: "demo"
  tools: [lookup]
  budget: { turns: 4, calls: 4, time: 30s }
}

agent answer(question) {
  system: "Use local docs."
  user: question
  tools: [lookup]
} flow {
  history := messages {
    system: system
    user: question
  }
  result, err := turn {
    model: model
    messages: history
  }
  return result, err
}

budget { turns: 1 } {
  result, err := answer("What is Leia?")
}
